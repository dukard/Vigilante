    private class android_to_ws20 extends AsyncTask<String, Object, Object > {

        String METHOD_NAME="";
        String SOAP_ACTION="http://tempuri.org/";
        final String NAMESPACE="http://tempuri.org/";
        final String URL="http://prospectos.carone.com.mx:8880/ws4/servicioweb.asmx";
        String idvigilante;

        public android_to_ws20(String metodo,String idvigilante) {
            this.METHOD_NAME=metodo;
            this.SOAP_ACTION += metodo;
            this.idvigilante=idvigilante;
        }

        @Override
        protected String doInBackground(String... params) {
            String resultado="";

            try {
                SoapObject request = new SoapObject(NAMESPACE, METHOD_NAME);
                request.addProperty("idvigilante", idvigilante);
                SoapSerializationEnvelope sobre = new SoapSerializationEnvelope(SoapEnvelope.VER11);
                new MarshalBase64().register(sobre);
                sobre.dotNet = true;
                sobre.setOutputSoapObject(request);
                HttpTransportSE transporte;
                transporte = new HttpTransportSE(URL);
                transporte.call(SOAP_ACTION, sobre);
                SoapPrimitive r = (SoapPrimitive) sobre.getResponse();
                resultado = r.toString();
            }
            catch (Exception e) {
                Toast.makeText(getActivity(), e.toString(), Toast.LENGTH_LONG).show();
                this.cancel(true);
            }

            return resultado;
        }

        @Override
        protected void onPostExecute(Object result) {
            try {
                if(result.toString().indexOf("Sin Datos") >= 0){
                    Toast.makeText(getActivity(), "Sin Datos", Toast.LENGTH_LONG).show();
                }
                else  if(result.toString().indexOf("Error:") >= 0){
                    Toast.makeText(getActivity(), result.toString(), Toast.LENGTH_LONG).show();
                }
                else if(result.toString().indexOf("Error") >= 0){
                    Toast.makeText(getActivity(), result.toString(), Toast.LENGTH_LONG).show();
                }
                else if(result.toString().compareTo("")==0){
                    Toast.makeText(getActivity(), result.toString(), Toast.LENGTH_LONG).show();
                }
                else{
                    pormientras(result.toString());
                }
                super.onPostExecute(result);
            }
            catch (Exception e) {
                Toast.makeText(getActivity(), e.toString(), Toast.LENGTH_LONG).show();
                this.cancel(true);
            }
        }
    }



    private boolean hayono(){
        ConnectivityManager cm =(ConnectivityManager) getActivity().getSystemService(Context.CONNECTIVITY_SERVICE);
        NetworkInfo netInfo = cm.getActiveNetworkInfo();
        if (netInfo != null && netInfo.isConnectedOrConnecting()&&netInfo.isAvailable()&&netInfo.isConnected()){
            try {
                process = Runtime.getRuntime().exec("/system/bin/ping -c 1 " + servidor);
                int mPingResult = process.waitFor();
                if(mPingResult == 0){
                    BufferedReader reader = new BufferedReader(new InputStreamReader(process.getInputStream()));
                    int i,l;
                    char[] buffer = new char[4096];
                    StringBuffer output = new StringBuffer();
                    while ((i = reader.read(buffer)) > 0)
                        output.append(buffer, 0, i);
                    reader.close();
                    STR = output.toString();
                    l=STR.indexOf("received,");
                    if(l>0){
                        l=Integer.parseInt((STR.substring((l-3),l)).trim());
                        if(l>0){
                            return true;
                        }
                        else{

                        }
                    }
                    else{
                        return false;
